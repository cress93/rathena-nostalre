// the way there, by the quest board
prontera,147,187,4	script	Expedition Guide#endgame	4_M_JOB_KNIGHT2,{
	mes "[Expedition Guide]";
	if (BaseLevel < 200) {
		mes "Past Midgard lie Muspelheim's fire and Isgard, on the World Serpent's back.";
		mes "Come back at level 200, and I will take you there.";
		close;
	}
	mes "Where to? The monsters there are as strong as anything in the world.";
	for (.@i = 0; .@i < getarraysize(.map$); .@i++)
		.@menu$ += .name$[.@i] + " (Lv " + .lv[.@i] + "):";
	.@i = select(.@menu$ + "Not now") - 1;
	if (.@i == getarraysize(.map$))
		close;
	close2;
	warp .map$[.@i], 0, 0;
	end;

OnInit:
	setarray .map$, "mu_fild01", "jor_tail", "jor_back1", "jor_back2", "jor_back3", "jor_back4", "jor_back5", "jor_twice", "jor_back6";
	setarray .name$, "Muspelheim", "Serpent's Tail", "Serpent's Back 1", "Serpent's Back 2", "Serpent's Back 3", "Serpent's Back 4", "Serpent's Back 5", "Serpent's Coil", "Serpent's Back 6";
	setarray .lv, 210, 208, 211, 212, 213, 220, 220, 222, 244;
	end;
